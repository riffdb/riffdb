// Review-only declarations. This file is not compiled.
// Declaration field order is JSON object key order. All enums serialize to the
// exact lowercase_snake_case spelling of the variant shown. Option::None omits
// the key; it never emits JSON null. Vec order is semantic producer order.

const OUTPUT_SCHEMA: &str = "riffdb.cli.output/v1";
const MAX_INPUT_BYTES: usize = 1_048_576;
const MAX_OUTPUT_MODEL_BYTES: usize = 4_194_304;
const MAX_RENDERED_BYTES: usize = 4_194_304;

// Custom envelope serialization emits one of these exact key orders.
struct SuccessEnvelope<T> {
    schema: Const<"riffdb.cli.output/v1">,
    command: CommandIdentity,
    ok: Const<true>,
    result: T,
}

struct ErrorEnvelope<E> {
    schema: Const<"riffdb.cli.output/v1">,
    command: CommandIdentity,
    ok: Const<false>,
    error: E,
}

    ContractValidate,     // "contract.validate"
    ContractDeploy,       // "contract.deploy"
    CommandExecute,       // "command.execute"
    CommandOutcome,       // "command.outcome"
    EntityGet,            // "entity.get"

    ServerHealth,         // "server.health"
    DemoBudget,           // "demo.budget"
}

// ---------- Structural scalar and Value DTOs ----------

type U64String = String;      // canonical 0|[1-9][0-9]*, bounded to u64
type I64String = String;      // canonical 0|-?[1-9][0-9]*, bounded to i64
type UuidText = String;       // canonical lowercase hyphenated UUID text
type UuidV7Text = String;     // UuidText additionally requiring UUIDv7/variant
type Hash256Hex = String;     // exactly 64 lowercase hexadecimal bytes
type PaddedBase64 = String;   // RFC 4648 standard alphabet with required padding

// Input and output are intentionally distinct recursive families. Submitted
// records may use ID, source name, or both; service-produced record output may
// use only stable field IDs. No output converter can emit an input-only name.
enum CliInputValue {
    Null { type: Const<"null"> },
    Bool { type: Const<"bool">, value: bool },
    I64 { type: Const<"i64">, value: I64String },
    U64 { type: Const<"u64">, value: U64String },
    Decimal {
        type: Const<"decimal">,
        coefficient_twos_complement: PaddedBase64,
        scale: u32,
    },
    Money {
        type: Const<"money">,
        currency: String,
        amount: CliDecimal,
    },
    String { type: Const<"string">, value: String },
    Bytes { type: Const<"bytes">, value: PaddedBase64 },
    Uuid { type: Const<"uuid">, value: UuidText },
    Date { type: Const<"date">, days_since_unix_epoch: i32 },
    Timestamp { type: Const<"timestamp">, seconds: I64String, nanos: u32 },
    Enum {
        type: Const<"enum">,
        type_id: NonZeroU32,
        variant_id: NonZeroU32,
        name: Option<String>,
    },
    List { type: Const<"list">, values: Vec<CliInputValue> },
    Record { type: Const<"record">, fields: Vec<CliInputField> },
}

enum CliOutputValue {
    Null { type: Const<"null"> },
    Bool { type: Const<"bool">, value: bool },
    I64 { type: Const<"i64">, value: I64String },
    U64 { type: Const<"u64">, value: U64String },
    Decimal {
        type: Const<"decimal">,
        coefficient_twos_complement: PaddedBase64,
        scale: u32,
    },
    Money {
        type: Const<"money">,
        currency: String,
        amount: CliDecimal,
    },
    String { type: Const<"string">, value: String },
    Bytes { type: Const<"bytes">, value: PaddedBase64 },
    Uuid { type: Const<"uuid">, value: UuidText },
    Date { type: Const<"date">, days_since_unix_epoch: i32 },
    Timestamp { type: Const<"timestamp">, seconds: I64String, nanos: u32 },
    Enum {
        type: Const<"enum">,
        type_id: NonZeroU32,
        variant_id: NonZeroU32,
        name: Option<String>,
    },
    List { type: Const<"list">, values: Vec<CliOutputValue> },
    Record { type: Const<"record">, fields: Vec<CliOutputField> },
}

struct CliDecimal {
    coefficient_twos_complement: PaddedBase64,
    scale: u32,
}

// Input preserves order. At least one identity is present; both must resolve to
// the same field. Output always uses CliOutputField instead.
struct CliInputField {
    field_id: Option<NonZeroU32>,
    name: Option<String>,
    value: CliInputValue,
}

struct CliRecordInput {
    type: Const<"record">,
    fields: Vec<CliInputField>,
}

// Service-produced canonical output is increasing by field_id and carries no
// source-name identity.
struct CliRecordOutput {
    fields: Vec<CliOutputField>,
}

struct CliOutputField {
    field_id: NonZeroU32,
    value: CliOutputValue,
}

struct TimestampDto { seconds: I64String, nanos: u32 }

// PublicError exact code/message/recovery registry:
// validation_failed | request validation failed | correct_request
// idempotency_key_reuse | idempotency key was reused with different input | correct_request
// authorization_denied | operation is not authorized | obtain_permission
// concurrency_deadline_exceeded | concurrency deadline exceeded | retry
// contract_mismatch | contract version or plan does not match | refresh_contract
// storage_unavailable | storage is temporarily unavailable | retry
// outcome_unknown | command outcome is not yet known | resolve_with_same_idempotency_key
// internal_defect | an internal error occurred | contact_operator
// command_execution_failed | command execution failed | contact_operator
